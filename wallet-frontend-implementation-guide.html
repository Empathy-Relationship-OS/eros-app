1. Token Purchase Flow

POST /wallet/purchase

Purchase a token package using Stripe payment

Request Payload

{
  "packageType": "STARTER" | "POPULAR" | "PREMIUM" | "MEGA",
  "paymentMethodId": "pm_xxxxxxxxxxxxx",  // From Stripe Elements
  "idempotencyKey": "uuid-v4-string",     // Client-generated
  "acceptedTerms": true                    // Must be true
}

Response

{
  "clientSecret": "pi_xxx_secret_xxx",     // Use with Stripe to confirm
  "paymentIntentId": "pi_xxxxxxxxxxxxx",   // Stripe payment intent ID
  "amount": 700,                           // Amount in pence
  "currency": "gbp",
  "tokenAmount": 1.0,                      // Tokens being purchased
  "status": "PENDING",                     // PENDING | COMPLETED | FAILED
  "newBalance": 6.0,                       // Optional: if completed
  "transactionId": 123,                    // Optional: if completed
  "acceptedTerms": true
}

Implementation Steps

  1. Setup Stripe SDK - Initialize with your publishable key
  2. Display Token Packages - Show available packages with pricing
  3. Terms Acceptance - Require user to accept terms and conditions
  4. Payment Method Collection - Use Stripe Elements for card input
  5. Generate Idempotency Key - Create UUID v4 client-side
  6. Call Backend - POST to /wallet/purchase
  7. Confirm Payment - Use clientSecret with Stripe SDK
  8. Handle Result - Update UI based on payment status

3. Create Purchase Service

import 'package:http/http.dart' as http;
import 'dart:convert';
import 'package:uuid/uuid.dart';

class WalletService {
  final String baseUrl = 'https://api.eros.app';
  final String firebaseToken;

  WalletService(this.firebaseToken);

  Future<PurchaseResponse> purchaseTokens({
    required String packageType,
    required String paymentMethodId,
    required bool acceptedTerms,
  }) async {
    // Generate idempotency key
    final idempotencyKey = Uuid().v4();

    final response = await http.post(
      Uri.parse('$baseUrl/wallet/purchase'),
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer $firebaseToken',
      },
      body: jsonEncode({
        'packageType': packageType,
        'paymentMethodId': paymentMethodId,
        'idempotencyKey': idempotencyKey,
        'acceptedTerms': acceptedTerms,
      }),
    );

    if (response.statusCode == 201) {
      return PurchaseResponse.fromJson(jsonDecode(response.body));
    } else {
      throw Exception('Failed to purchase tokens: ${response.body}');
    }
  }
}

4. Create Purchase UI Widget

class TokenPurchaseScreen extends StatefulWidget {
  @override
  _TokenPurchaseScreenState createState() => _TokenPurchaseScreenState();
}

class _TokenPurchaseScreenState extends State<TokenPurchaseScreen> {
  String? selectedPackage;
  bool acceptedTerms = false;
  bool isProcessing = false;

  final packages = [
    {'type': 'STARTER', 'tokens': 1, 'price': 7.00},
    {'type': 'POPULAR', 'tokens': 3, 'price': 20.00},
    {'type': 'PREMIUM', 'tokens': 5, 'price': 32.50},
    {'type': 'MEGA', 'tokens': 10, 'price': 45.00},
  ];

  Future<void> handlePurchase() async {
    if (selectedPackage == null || !acceptedTerms) return;

    setState(() => isProcessing = true);

    try {
      // 1. Create payment method
      final paymentMethod = await Stripe.instance.createPaymentMethod(
        params: PaymentMethodParams.card(
          paymentMethodData: PaymentMethodData(),
        ),
      );

      // 2. Call backend to create payment intent
      final walletService = WalletService(firebaseToken);
      final purchase = await walletService.purchaseTokens(
        packageType: selectedPackage!,
        paymentMethodId: paymentMethod.id,
        acceptedTerms: acceptedTerms,
      );

      // 3. Confirm payment with Stripe
      final paymentIntent = await Stripe.instance.confirmPayment(
        paymentIntentClientSecret: purchase.clientSecret,
      );

      // 4. Handle success
      if (paymentIntent.status == PaymentIntentsStatus.Succeeded) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('✅ Purchase successful!')),
        );
        // Navigate to wallet or refresh balance
      }
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('❌ Purchase failed: $e')),
      );
    } finally {
      setState(() => isProcessing = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Purchase Tokens')),
      body: Padding(
        padding: EdgeInsets.all(16),
        child: Column(
          children: [
            // Package selection
            ...packages.map((pkg) => PackageCard(
              type: pkg['type'] as String,
              tokens: pkg['tokens'] as int,
              price: pkg['price'] as double,
              isSelected: selectedPackage == pkg['type'],
              onTap: () => setState(() => selectedPackage = pkg['type'] as String),
            )),

            // Terms checkbox
            CheckboxListTile(
              title: Text('I accept the terms and conditions'),
              value: acceptedTerms,
              onChanged: (val) => setState(() => acceptedTerms = val ?? false),
            ),

            // Card input
            CardField(
              onCardChanged: (card) {},
            ),

            SizedBox(height: 20),

            // Purchase button
            ElevatedButton(
              onPressed: (selectedPackage != null && acceptedTerms && !isProcessing)
                  ? handlePurchase
                  : null,
              child: isProcessing
                  ? CircularProgressIndicator()
                  : Text('Purchase'),
            ),
          ],
        ),
      ),
    );
  }
}

Transaction Types

  • PURCHASE - Token purchase via Stripe (amount is positive)
  • SPEND - Date commitment (amount is negative)
  • REFUND - Cancelled date refund (amount is positive)
  • ADJUSTMENT - Manual admin adjustment

Transaction Statuses

  • PENDING - Payment intent created, awaiting confirmation
  • COMPLETED - Successfully processed
  • FAILED - Payment/transaction failed
  • CANCELLED - Payment cancelled
  • REFUNDED - Payment refunded
  • REFUND_FAILED - Refund attempt failed

class TransactionHistoryScreen extends StatefulWidget {
  @override
  _TransactionHistoryScreenState createState() => _TransactionHistoryScreenState();
}

class _TransactionHistoryScreenState extends State<TransactionHistoryScreen> {
  List<Transaction> transactions = [];
  bool isLoading = false;
  bool hasMore = true;
  int offset = 0;
  final int limit = 20;
  String? filterType;

  ScrollController scrollController = ScrollController();

  @override
  void initState() {
    super.initState();
    loadTransactions();

    // Infinite scroll
    scrollController.addListener(() {
      if (scrollController.position.pixels ==
          scrollController.position.maxScrollExtent &&
          hasMore &&
          !isLoading) {
        loadTransactions();
      }
    });
  }

  Future<void> loadTransactions() async {
    if (isLoading) return;

    setState(() => isLoading = true);

    try {
      final response = await http.get(
        Uri.parse('$baseUrl/wallet/transactions')
            .replace(queryParameters: {
              'limit': limit.toString(),
              'offset': offset.toString(),
              if (filterType != null) 'type': filterType!,
            }),
        headers: {'Authorization': 'Bearer $firebaseToken'},
      );

      if (response.statusCode == 200) {
        final data = TransactionHistory.fromJson(jsonDecode(response.body));
        setState(() {
          transactions.addAll(data.transactions);
          offset += limit;
          hasMore = data.hasMore;
        });
      }
    } catch (e) {
      print('Error loading transactions: $e');
    } finally {
      setState(() => isLoading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text('Transaction History'),
        actions: [
          // Filter dropdown
          PopupMenuButton<String>(
            onSelected: (type) {
              setState(() {
                filterType = type;
                transactions.clear();
                offset = 0;
                hasMore = true;
              });
              loadTransactions();
            },
            itemBuilder: (context) => [
              PopupMenuItem(value: null, child: Text('All')),
              PopupMenuItem(value: 'PURCHASE', child: Text('Purchases')),
              PopupMenuItem(value: 'SPEND', child: Text('Spends')),
              PopupMenuItem(value: 'REFUND', child: Text('Refunds')),
            ],
          ),
        ],
      ),
      body: ListView.builder(
        controller: scrollController,
        itemCount: transactions.length + (hasMore ? 1 : 0),
        itemBuilder: (context, index) {
          if (index == transactions.length) {
            return Center(child: CircularProgressIndicator());
          }

          final tx = transactions[index];
          return TransactionTile(transaction: tx);
        },
      ),
    );
  }
}

class TransactionTile extends StatelessWidget {
  final Transaction transaction;

  @override
  Widget build(BuildContext context) {
    final isCredit = transaction.amount > 0;
    final icon = _getIcon(transaction.type);
    final color = _getColor(transaction.type, isCredit);

    return ListTile(
      leading: CircleAvatar(
        backgroundColor: color.withOpacity(0.2),
        child: Icon(icon, color: color),
      ),
      title: Text(transaction.description),
      subtitle: Text(
        '${_formatDate(transaction.createdAt)} • ${transaction.status}',
      ),
      trailing: Column(
        mainAxisAlignment: MainAxisAlignment.center,
        crossAxisAlignment: CrossAxisAlignment.end,
        children: [
          Text(
            '${isCredit ? '+' : ''}${transaction.amount} tokens',
            style: TextStyle(
              fontWeight: FontWeight.bold,
              color: color,
              fontSize: 16,
            ),
          ),
          if (transaction.amountPaid != null)
            Text(
              '£${(transaction.amountPaid! / 100).toStringAsFixed(2)}',
              style: TextStyle(fontSize: 12, color: Colors.grey),
            ),
        ],
      ),
      onTap: () {
        // Navigate to transaction details or related date
        if (transaction.relatedDateId != null) {
          Navigator.push(
            context,
            MaterialPageRoute(
              builder: (_) => DateDetailsScreen(
                dateId: transaction.relatedDateId!,
              ),
            ),
          );
        }
      },
    );
  }

  IconData _getIcon(String type) {
    switch (type) {
      case 'PURCHASE': return Icons.add_shopping_cart;
      case 'SPEND': return Icons.favorite;
      case 'REFUND': return Icons.refresh;
      default: return Icons.swap_horiz;
    }
  }

  Color _getColor(String type, bool isCredit) {
    switch (type) {
      case 'PURCHASE': return Colors.green;
      case 'SPEND': return Colors.red;
      case 'REFUND': return Colors.orange;
      default: return Colors.blue;
    }
  }

  String _formatDate(String iso) {
    final date = DateTime.parse(iso);
    return '${date.day}/${date.month}/${date.year}';
  }
}

class DateCommitWidget extends StatefulWidget {

  final String activity;
  final double tokenCost;

  @override
  _DateCommitWidgetState createState() => _DateCommitWidgetState();
}

class _DateCommitWidgetState extends State<DateCommitWidget> {
  bool isCommitting = false;

  Future<void> commitToDate() async {
    setState(() => isCommitting = true);

    try {
      final idempotencyKey = Uuid().v4();

      final response = await http.post(
        Uri.parse('$baseUrl/wallet/spend'),
        headers: {
          'Content-Type': 'application/json',
          'Authorization': 'Bearer $firebaseToken',
        },
        body: jsonEncode({
          'relatedDateId': widget.dateId,
          'activity': widget.activity,
          'idempotencyKey': idempotencyKey,
        }),
      );

      if (response.statusCode == 201) {
        final transaction = Transaction.fromJson(jsonDecode(response.body));

        // Show success
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text('✅ Committed! New balance: ${transaction.balanceAfter}'),
            backgroundColor: Colors.green,
          ),
        );

        // Navigate to date confirmation screen
        Navigator.push(
          context,
          MaterialPageRoute(
            builder: (_) => DateConfirmationScreen(dateId: widget.dateId),
          ),
        );
      } else if (response.statusCode == 400) {
        // Insufficient balance or already paid
        final error = jsonDecode(response.body);
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('❌ ${error['message']}')),
        );
      }
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('❌ Error: $e')),
      );
    } finally {
      setState(() => isCommitting = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return ElevatedButton(
      onPressed: isCommitting ? null : commitToDate,
      child: isCommitting
          ? CircularProgressIndicator()
          : Text('Commit (${widget.tokenCost} tokens)'),
    );
  }
}

    }
  }

      return SizedBox.shrink();
    }

    return TextButton(
      onPressed: isRefunding ? null : requestRefund,
      child: isRefunding
          ? SizedBox(
              width: 16,
              height: 16,
              child: CircularProgressIndicator(strokeWidth: 2),
            )
          : Text('Request Refund'),
    );
  }

⚙️ Key Considerations

1. Error Handling

Status Code Meaning Example
200 OK Success (GET requests) Balance retrieved
201 Created Success (POST requests) Purchase created
400 Bad Request Invalid input Invalid package type, insufficient balance
401 Unauthorized Missing/invalid auth token Firebase token expired
404 Not Found Resource not found Wallet or transaction doesn't exist
409 Conflict Conflict with current state Transaction already refunded

2. Idempotency Keys

Critical for preventing duplicate charges:
  • Generate UUID v4 client-side for each operation
  • Store in local state during payment flow
  • Reuse same key if user retries failed operation
  • Backend returns existing transaction for duplicate keys
  • Use different keys for different purchases/operations

🎯 Idempotency Key Management

class IdempotencyManager {
  final _storage = GetStorage(); // Or SharedPreferences

  String getOrCreateKey(String operationId) {
    final key = 'idempotency_$operationId';

    if (_storage.hasData(key)) {
      return _storage.read(key);
    }

    final newKey = Uuid().v4();
    _storage.write(key, newKey);
    return newKey;
  }

  void clearKey(String operationId) {
    _storage.remove('idempotency_$operationId');
  }
}

// Usage
final manager = IdempotencyManager();

// When starting a purchase
final purchaseId = 'purchase_${packageType}_${timestamp}';
final idempotencyKey = manager.getOrCreateKey(purchaseId);

// After successful completion
manager.clearKey(purchaseId);

3. Currency Handling

  • All prices are in GBP
  • API returns amounts in pence (smallest unit)
  • Divide by 100 to display in pounds: £{amount / 100}
  • Token amounts are decimal values (e.g., 1.0, 3.0, 5.0)

4. Stripe Integration Flow

Two-step payment process:
  1. Backend creates payment intent → returns clientSecret
  2. Frontend confirms payment with Stripe using clientSecret

This separation ensures PCI compliance - card details never touch your backend.

5. State Management

  • Track payment status: PENDING → COMPLETED/FAILED
  • Update balance immediately after successful operations
  • Show pending transactions differently in UI
  • Implement optimistic updates with rollback on failure

6. Security Best Practices

  • Never expose Stripe secret key in frontend code
  • Always use HTTPS for API calls
  • Validate Firebase token expiry client-side
  • Refresh Firebase token before making API calls if close to expiry
  • Store sensitive data (tokens, keys) securely using platform keychain

import 'package:http/http.dart' as http;
import 'dart:convert';
import 'package:uuid/uuid.dart';

class WalletService {
  final String baseUrl;
  final String firebaseToken;

  WalletService({
    required this.baseUrl,
    required this.firebaseToken,
  });

  Map<String, String> get _headers => {
    'Content-Type': 'application/json',
    'Authorization': 'Bearer $firebaseToken',
  };

  // Get wallet balance
  Future<WalletBalance> getBalance() async {
    final response = await http.get(
      Uri.parse('$baseUrl/wallet/balance'),
      headers: _headers,
    );

    if (response.statusCode == 200) {
      return WalletBalance.fromJson(jsonDecode(response.body));
    } else {
      throw WalletException('Failed to fetch balance', response.statusCode);
    }
  }

  // Get transaction history
  Future<TransactionHistory> getTransactions({
    required int limit,
    required int offset,
    String? type,
  }) async {
    final uri = Uri.parse('$baseUrl/wallet/transactions').replace(
      queryParameters: {
        'limit': limit.toString(),
        'offset': offset.toString(),
        if (type != null) 'type': type,
      },
    );

    final response = await http.get(uri, headers: _headers);

    if (response.statusCode == 200) {
      return TransactionHistory.fromJson(jsonDecode(response.body));
    } else {
      throw WalletException('Failed to fetch transactions', response.statusCode);
    }
  }

  // Purchase tokens
  Future<PurchaseResponse> purchaseTokens({
    required String packageType,
    required String paymentMethodId,
    required bool acceptedTerms,
    String? customIdempotencyKey,
  }) async {
    final idempotencyKey = customIdempotencyKey ?? Uuid().v4();

    final response = await http.post(
      Uri.parse('$baseUrl/wallet/purchase'),
      headers: _headers,
      body: jsonEncode({
        'packageType': packageType,
        'paymentMethodId': paymentMethodId,
        'idempotencyKey': idempotencyKey,
        'acceptedTerms': acceptedTerms,
      }),
    );

    if (response.statusCode == 201) {
      return PurchaseResponse.fromJson(jsonDecode(response.body));
    } else {
      final error = jsonDecode(response.body);
      throw WalletException(
        error['message'] ?? 'Purchase failed',
        response.statusCode,
      );
    }
  }

  // Spend tokens
  Future<Transaction> spendTokens({
    required int relatedDateId,
    required String activity,
    String? customIdempotencyKey,
  }) async {
    final idempotencyKey = customIdempotencyKey ?? Uuid().v4();

    final response = await http.post(
      Uri.parse('$baseUrl/wallet/spend'),
      headers: _headers,
      body: jsonEncode({
        'relatedDateId': relatedDateId,
        'activity': activity,
        'idempotencyKey': idempotencyKey,
      }),
    );

    if (response.statusCode == 201) {
      return Transaction.fromJson(jsonDecode(response.body));
    } else {
      final error = jsonDecode(response.body);
      throw WalletException(
        error['message'] ?? 'Spend failed',
        response.statusCode,
      );
    }
  }

  // Refund tokens
  Future<RefundResponse> refundTokens({
    required int transactionId,
    required String stripePaymentIntent,
    String? customIdempotencyKey,
  }) async {
    final idempotencyKey = customIdempotencyKey ?? Uuid().v4();

    final response = await http.post(
      Uri.parse('$baseUrl/wallet/refund'),
      headers: _headers,
      body: jsonEncode({
        'transactionId': transactionId,
        'stripePaymentIntent': stripePaymentIntent,
        'idempotencyKey': idempotencyKey,
      }),
    );

    if (response.statusCode == 201) {
      return RefundResponse.fromJson(jsonDecode(response.body));
    } else {
      final error = jsonDecode(response.body);
      throw WalletException(
        error['message'] ?? 'Refund failed',
        response.statusCode,
      );
    }
  }
